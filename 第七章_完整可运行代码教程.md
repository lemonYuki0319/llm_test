### 3.3 大模型接口封装（utils/llm_client.py）

统一封装多模型API调用，支持通义千问、OpenAI等，提供流式输出、Token计数等功能。

```python
import time
import json
import tiktoken
from openai import OpenAI
from config.settings import (
    DASHSCOPE_API_KEY, DASHSCOPE_BASE_URL,
    OPENAI_API_KEY, OPENAI_BASE_URL,
    DEFAULT_MODEL, DEFAULT_TEMPERATURE, DEFAULT_MAX_TOKENS, DEFAULT_TIMEOUT,
    MODELS
)


class LLMClient:
    def __init__(self, model=DEFAULT_MODEL):
        self.model = model
        provider = MODELS.get(model, {}).get("provider", "dashscope")
        if provider == "dashscope":
            self.client = OpenAI(
                api_key=DASHSCOPE_API_KEY,
                base_url=DASHSCOPE_BASE_URL
            )
        else:
            self.client = OpenAI(
                api_key=OPENAI_API_KEY,
                base_url=OPENAI_BASE_URL
            )

    def chat(self, messages, temperature=DEFAULT_TEMPERATURE,
             max_tokens=DEFAULT_MAX_TOKENS, timeout=DEFAULT_TIMEOUT):
        start_time = time.time()
        response = self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            temperature=temperature,
            max_tokens=max_tokens,
            timeout=timeout
        )
        latency = time.time() - start_time
        content = response.choices[0].message.content
        usage = response.usage
        return {
            "content": content,
            "model": response.model,
            "latency": latency,
            "prompt_tokens": usage.prompt_tokens if usage else 0,
            "completion_tokens": usage.completion_tokens if usage else 0,
            "total_tokens": usage.total_tokens if usage else 0,
        }

    def simple_chat(self, question, system_prompt=None, temperature=DEFAULT_TEMPERATURE):
        messages = []
        if system_prompt:
            messages.append({"role": "system", "content": system_prompt})
        messages.append({"role": "user", "content": question})
        return self.chat(messages, temperature=temperature)

    def chat_stream(self, messages, temperature=DEFAULT_TEMPERATURE,
                    max_tokens=DEFAULT_MAX_TOKENS):
        ttft = None
        start_time = time.time()
        stream = self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            temperature=temperature,
            max_tokens=max_tokens,
            stream=True
        )
        chunks = []
        token_times = []
        prev_time = None
        for chunk in stream:
            if chunk.choices[0].delta.content:
                current_time = time.time()
                if ttft is None:
                    ttft = current_time - start_time
                if prev_time:
                    token_times.append(current_time - prev_time)
                prev_time = current_time
                chunks.append(chunk.choices[0].delta.content)

        full_content = "".join(chunks)
        avg_tpot = (sum(token_times) / len(token_times) * 1000) if token_times else 0
        return {
            "content": full_content,
            "ttft": ttft,
            "avg_tpot_ms": avg_tpot,
            "token_count": len(token_times) + 1,
            "total_latency": time.time() - start_time,
        }

    @staticmethod
    def count_tokens(text, model="cl100k_base"):
        try:
            encoding = tiktoken.get_encoding(model)
            return len(encoding.encode(text))
        except Exception:
            return len(text) // 4


def call_llm(question, model=DEFAULT_MODEL, system_prompt=None, temperature=DEFAULT_TEMPERATURE):
    client = LLMClient(model=model)
    result = client.simple_chat(question, system_prompt=system_prompt, temperature=temperature)
    return result["content"]


def call_llm_with_messages(messages, model=DEFAULT_MODEL, temperature=DEFAULT_TEMPERATURE):
    client = LLMClient(model=model)
    result = client.chat(messages, temperature=temperature)
    return result["content"]
```

### 3.4 RAG链路辅助工具（utils/rag_helper.py）

封装RAG检索和问答链路，便于测试验证。

```python
from langchain_community.vectorstores import FAISS
from langchain_community.embeddings import DashScopeEmbeddings
from langchain.chains import RetrievalQA
from langchain_community.llms import Tongyi
from config.settings import DASHSCOPE_API_KEY


class RAGHelper:
    def __init__(self, vectorstore_path="knowledge_base"):
        self.embeddings = DashScopeEmbeddings(
            model="text-embedding-v2",
            dashscope_api_key=DASHSCOPE_API_KEY
        )
        self.vectorstore_path = vectorstore_path
        self._vectorstore = None
        self._qa_chain = None

    @property
    def vectorstore(self):
        if self._vectorstore is None:
            self._vectorstore = FAISS.load_local(
                self.vectorstore_path, self.embeddings,
                allow_dangerous_deserialization=True
            )
        return self._vectorstore

    @property
    def qa_chain(self):
        if self._qa_chain is None:
            llm = Tongyi(dashscope_api_key=DASHSCOPE_API_KEY)
            self._qa_chain = RetrievalQA.from_chain_type(
                llm=llm,
                chain_type="stuff",
                retriever=self.vectorstore.as_retriever(search_kwargs={"k": 3}),
                return_source_documents=True
            )
        return self._qa_chain

    def search(self, query, k=3):
        docs = self.vectorstore.similarity_search(query, k=k)
        return docs

    def ask(self, question):
        result = self.qa_chain({"query": question})
        return {
            "answer": result["result"],
            "source_documents": result.get("source_documents", [])
        }

    def add_document(self, text, metadata=None):
        self.vectorstore.add_texts([text], metadatas=[metadata or {}])
        self.vectorstore.save_local(self.vectorstore_path)

    def rebuild_index(self):
        self._vectorstore = None
        self._qa_chain = None
```

### 3.5 自定义断言方法（utils/assertion_helper.py）

封装AI测试特有的断言方法，支持关键词包含、语义相似度、JSON格式校验等。

```python
import json
import re
from bert_score import score as bert_score_fn
from rouge_score import rouge_scorer
from nltk.translate.bleu_score import sentence_bleu


def assert_keywords_in_response(response, keywords, msg=None):
    missing = [kw for kw in keywords if kw not in response]
    assert not missing, msg or f"回答中缺少关键词：{missing}"


def assert_no_keywords_in_response(response, keywords, msg=None):
    found = [kw for kw in keywords if kw in response]
    assert not found, msg or f"回答中包含不应出现的关键词：{found}"


def assert_json_format(response, required_fields=None, msg=None):
    try:
        result = json.loads(response)
    except json.JSONDecodeError:
        assert False, msg or "输出不是有效的JSON格式"
    if required_fields:
        missing = [f for f in required_fields if f not in result]
        assert not missing, msg or f"JSON中缺少必要字段：{missing}"
    return result


def assert_bertscore_above(candidate, reference, threshold=0.7, lang="zh", msg=None):
    P, R, F1 = bert_score_fn([candidate], [reference], lang=lang)
    score = F1[0].item()
    assert score >= threshold, msg or f"BERTScore F1={score:.4f}低于阈值{threshold}"
    return score


def assert_rouge_above(candidate, reference, threshold=0.5, msg=None):
    scorer = rouge_scorer.RougeScorer(["rouge1", "rougeL"], use_stemmer=False)
    scores = scorer.score(reference, candidate)
    rouge1 = scores["rouge1"].fmeasure
    assert rouge1 >= threshold, msg or f"ROUGE-1={rouge1:.4f}低于阈值{threshold}"
    return rouge1


def assert_bleu_above(candidate_tokens, reference_tokens_list, threshold=0.3, msg=None):
    score = sentence_bleu(reference_tokens_list, candidate_tokens)
    assert score >= threshold, msg or f"BLEU={score:.4f}低于阈值{threshold}"
    return score


def assert_response_latency(latency, threshold=5.0, msg=None):
    assert latency <= threshold, msg or f"响应时间{latency:.2f}s超过阈值{threshold}s"


def assert_no_hallucination(response, context, msg=None):
    sentences = re.split(r'[。！？\n]', response)
    unsupported = []
    for sent in sentences:
        sent = sent.strip()
        if len(sent) < 5:
            continue
        has_support = any(char in context for char in sent[:10])
        if not has_support and len(sent) > 10:
            unsupported.append(sent)
    if unsupported:
        assert False, msg or f"以下内容可能为幻觉（无上下文支撑）：{unsupported[:3]}"
```

### 5.1 功能测试（testcases/test_functional.py）

```python
import pytest
from utils.llm_client import call_llm
from utils.assertion_helper import assert_keywords_in_response
from config.prompts import SYSTEM_PROMPTS


class TestFunctional:
    @pytest.mark.parametrize("item_index", range(5))
    def test_accuracy_with_golden_set(self, llm_client, golden_set, item_index):
        item = golden_set[item_index]
        result = llm_client.simple_chat(
            item["question"],
            system_prompt=SYSTEM_PROMPTS["customer_service"]
        )
        assert_keywords_in_response(result["content"], item["keywords"])

    def test_context_coherence(self, llm_client):
        messages = [
            {"role": "user", "content": "我想买一台笔记本电脑"},
            {"role": "assistant", "content": "好的，请问您的预算是多少？"},
            {"role": "user", "content": "5000左右"},
            {"role": "assistant", "content": "推荐以下几款..."},
            {"role": "user", "content": "第一款能打游戏吗？"},
        ]
        result = llm_client.chat(messages)
```

### 5.3 性能测试（testcases/test_performance.py）

```python
import time
import pytest
from concurrent.futures import ThreadPoolExecutor, as_completed
from utils.llm_client import LLMClient
from config.prompts import SYSTEM_PROMPTS


class TestPerformance:
    def test_response_latency(self, llm_client):
        latencies = []
        for _ in range(10):
            result = llm_client.simple_chat(
                "退货政策是什么？",
                system_prompt=SYSTEM_PROMPTS["customer_service"]
            )
            latencies.append(result["latency"])

        avg_latency = sum(latencies) / len(latencies)
        p95_latency = sorted(latencies)[int(len(latencies) * 0.95)]
        max_latency = max(latencies)

        print(f"响应延迟 - 平均: {avg_latency:.2f}s, P95: {p95_latency:.2f}s, 最大: {max_latency:.2f}s")
        assert avg_latency < 5.0, f"平均响应时间{avg_latency:.2f}s超过5s阈值"

    def test_ttft(self, llm_client):
        messages = [
            {"role": "system", "content": SYSTEM_PROMPTS["customer_service"]},
            {"role": "user", "content": "退货政策是什么？"}
        ]
        result = llm_client.chat_stream(messages)
        print(f"TTFT: {result['ttft']:.3f}s, TPOT: {result['avg_tpot_ms']:.1f}ms/Token")
        assert result["ttft"] < 2.0, f"TTFT {result['ttft']:.3f}s超过2s阈值"

    def test_tpot(self, llm_client):
        messages = [
            {"role": "system", "content": SYSTEM_PROMPTS["customer_service"]},
            {"role": "user", "content": "详细说明退货流程"}
        ]
        result = llm_client.chat_stream(messages)
        print(f"TPOT: {result['avg_tpot_ms']:.1f}ms/Token, 总Token数: {result['token_count']}")
        assert result["avg_tpot_ms"] < 150, f"TPOT {result['avg_tpot_ms']:.1f}ms超过150ms阈值"

    def test_concurrent_requests(self):
        def send_request(_):
            client = LLMClient()
            try:
                result = client.simple_chat("退货政策是什么？")
                return {"success": True, "latency": result["latency"]}
            except Exception as e:
                return {"success": False, "error": str(e)}

        with ThreadPoolExecutor(max_workers=10) as executor:
            futures = [executor.submit(send_request, i) for i in range(20)]
            results = [f.result() for f in as_completed(futures)]

        success_count = sum(1 for r in results if r["success"])
        success_rate = success_count / len(results)
        print(f"并发测试 - 成功率: {success_rate:.0%}, 成功: {success_count}/{len(results)}")
        assert success_rate >= 0.90, f"并发成功率{success_rate:.0%}低于90%阈值"
```

```python
import json
```

```python
            with open("test_results/baseline_accuracy.json", "r") as f:
                baseline = json.load(f)
        except FileNotFoundError:

        print(f"基线准确率: {baseline['accuracy_rate']:.2%}, "
              f"当前准确率: {current_accuracy:.2%}, 漂移: {drift:.2%}")

            "date": datetime.date.today().isoformat(),
            "accuracy_rate": current_accuracy,
            "baseline_accuracy": baseline["accuracy_rate"],
            "drift": drift,
        }

        import os
        os.makedirs("test_results", exist_ok=True)
        with open(f"test_results/drift_report_{datetime.date.today().isoformat()}.json", "w") as f:
            json.dump(report, f, indent=2, ensure_ascii=False)

        assert drift <= 0.05, f"性能退化超过5%阈值，当前漂移{drift:.2%}"
```

```python
import json
import sys
from config.settings import QUALITY_THRESHOLDS


def check_quality_threshold():
    try:
        with open("test_results/quality_report.json", "r") as f:
            report = json.load(f)
    except FileNotFoundError:
        print("WARNING: quality_report.json not found, skipping threshold check")
        return

    blocked = False
    for metric, threshold in QUALITY_THRESHOLDS.items():
        actual = report.get(metric, 0)
        if metric == "hallucination_rate":
            if actual > threshold:
                print(f"BLOCKED: {metric} = {actual:.2%} > threshold {threshold:.2%}")
                blocked = True
        elif metric in ["avg_ttft", "avg_tpot_ms"]:
            if actual > threshold:
                print(f"BLOCKED: {metric} = {actual} > threshold {threshold}")
                blocked = True
        else:
            if actual < threshold:
                print(f"BLOCKED: {metric} = {actual:.2%} < threshold {threshold:.2%}")
                blocked = True

    if blocked:
        sys.exit(1)
    else:
        print("All quality thresholds passed!")


if __name__ == "__main__":
    check_quality_threshold()
```
